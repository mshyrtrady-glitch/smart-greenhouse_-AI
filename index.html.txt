<script>
function updateSystem() {
    let temp = document.getElementById('tempRange').value;
    let soil = document.getElementById('soilRange').value;
    let hum = document.getElementById('humRange').value;

    document.getElementById('tempVal').innerText = temp;
    document.getElementById('soilVal').innerText = soil;
    document.getElementById('humVal').innerText = hum;

    let pump = document.getElementById('pumpStatus');
    let fan = document.getElementById('fanStatus');
    let aiStatus = document.getElementById('aiDiseaseStatus');

    // منطق رطوبة التربة والمضخة
    if (soil < 30) {
        pump.className = "status-box status-danger";
        pump.innerText = "مضخة المياه: تعمل الآن! (التربة جافة < 30%)";
    } else {
        pump.className = "status-box status-safe";
        pump.innerText = "مضخة المياه: متوقفة (التربة في المعدل الطبيعي)";
    }

    // منطق الحرارة والمروحة
    if (temp > 32) {
        fan.className = "status-box status-danger";
        fan.innerText = "مروحة التهوية: تعمل بأقصى سرعة! (حرارة عالية > 32°C)";
    } else {
        fan.className = "status-box status-safe";
        fan.innerText = "مروحة التهوية: متوقفة (درجة الحرارة مستقرة)";
    }

    // نموذج محاكاة تحليل الأمراض بناءً على الظروف البيئية
    if (hum > 80 && temp > 28) {
        aiStatus.className = "status-box status-danger";
        aiStatus.innerText = "⚠️ تحذير AI: تم اكتشاف أعراض 'اللفحة المتأخرة' (Late Blight) بسبب ارتفاع الرطوبة والحرارة!";
    } else if (soil < 20) {
        aiStatus.className = "status-box status-danger";
        aiStatus.innerText = "⚠️ تحذير AI: النبات يعاني من 'إجهاد مائي حاد' (Severe Water Stress)!";
    } else {
        aiStatus.className = "status-box status-safe";
        aiStatus.innerText = "✅ حالة النبات: سليمة تماماً (Healthy Leaf - Confidence: 97.8%)";
    }
}
</script>
